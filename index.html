/**
 * Official GenLayerSpinner Component
 * Requires Tailwind CSS + custom keyframes.
 * SVG paths strictly map to the GenLayer Delta.
 */
import React from 'react';

export const GenLayerSpinner = ({ variant = 'vortex', size = 24 }) => {
  
  if (variant === 'micro') {
    return (
      <svg viewBox="0 0 40 40" width={size} height={size}>
        <defs>
          <linearGradient id="gl-g" x1="0" y1="0" x2="1" y2="1">
            <stop offset="0%" stopColor="#7A40FF" />
            <stop offset="100%" stopColor="#00F5D4" />
          </linearGradient>
        </defs>
        <g transform="scale(0.24) translate(34, 34)" fill="currentColor" opacity="0.3">
          <path d="M 46,10 L 10,90 L 46,78 L 30,62 L 46,46 Z" />
          <path d="M 54,10 L 90,90 L 54,78 L 70,62 L 54,46 Z" />
          <path d="M 50,50 L 62,62 L 50,74 L 38,62 Z" />
        </g>
        <circle cx="20" cy="20" r="18" fill="none" stroke="url(#gl-g)" 
          strokeWidth="4" strokeDasharray="80 30" strokeLinecap="round" 
          style={{ transformOrigin: 'center', animation: 'spin 1s linear infinite' }} 
        />
      </svg>
    );
  }
  
  // Other variants (Vortex, Apex) ...
  return null;
};
