/**
 * GenLayerSpinner.tsx
 * Official GenLayer Portal loading spinner component.
 * Uses exact geometric paths from the GenLayer logo.
 */
import React, { useId, useMemo } from "react";

export function GenLayerSpinner({
  variant = "orbit",
  size = 64,
  colorFrom = "#7A40FF",
  colorTo = "#00F5D4",
  speed = 1.6
}) {
  const uid = useId().replace(/:/g, "");
  const style = useMemo(() => ({
    "--gl-a": colorFrom,
    "--gl-b": colorTo,
    "--gl-duration": `${speed}s`,
    width: size, height: size
  }), [colorFrom, colorTo, speed, size]);

  return (
    <span style={style} className="gl-spinner inline-flex leading-none">
      {variant === "orbit" && <OrbitSVG uid={uid} />}
      {variant === "neural" && <NeuralSVG uid={uid} />}
      {variant === "portal" && <PortalSVG uid={uid} />}
    </span>
  );
}

function OrbitSVG({ uid }) {
  return (
    <svg viewBox="0 0 100 100" className="block overflow-visible w-full h-full">
      {`/* ... (SVG defs and gradients) ... */`}
      <g transform="translate(30, 30) scale(0.4)" fill="currentColor" opacity="0.15">
        <path d="M 46,10 L 10,90 L 46,78 L 30,62 L 46,46 Z" />
        <path d="M 54,10 L 90,90 L 54,78 L 70,62 L 54,46 Z" />
      </g>
      <g className="gl-orbit-outer">
        <circle cx="50" cy="50" r="42" fill="none" stroke={`url(#gl-oa)`} strokeWidth="6" strokeLinecap="round" strokeDasharray="198 66"/>
      </g>
      <g className="gl-orbit-inner">
        <circle cx="50" cy="50" r="29" fill="none" stroke={`url(#gl-ob)`} strokeWidth="5" strokeLinecap="round" strokeDasharray="109 73"/>
      </g>
      <g className="gl-orbit-core">
        <path d="M 50,50 L 62,62 L 50,74 L 38,62 Z" fill="var(--gl-b)" transform="translate(30, 30) scale(0.4)" />
      </g>
    </svg>
  );
}
